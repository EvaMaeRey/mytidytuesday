}

StatPost <- ggproto("StatPost", Stat, 
                    compute_group = compute_group_post,
                    required_aes = c("x", "y"))

geom_post <- make_constructor(GeomSegment, stat = StatPost)


library(ggplot2)

shot_fun <- function(x, v0 = 15, angle_deg = 30, g = 9.81, y0 = 1.7) {
  
  radians <- angle_deg * pi / 180
  
  y <- y0 + 
    x * tan(radians) - 
    (g * x ^ 2) / (2 * v0 ^ 2 * cos(radians) ^ 2)
  
  ifelse(y >= 0, y, 0)
  
}
