translate_unique <- function(from = c("peacock", "dolphin", "dolphin", "bat", "bat", "dog"), 
                             lookup = data.frame(id1 = c("peacock", "dolphin", "bat"),
                                                 id_esp = c("pavo real", "delfín", "murciélago"),
                                                 id3 = c(   NA,    NA, "bunny"),
                                                 .to = c("blue", "turquoise", "black")), 
                             to_column = .to, 
                             missing_value = "grey" |> alpha(.5)){
  
  from_unique <- unique(from)
  
  lookup_long <- lookup |> 
    pivot_longer(cols = -{{to_column}})
  
  data.frame(value = from) |>
    left_join(lookup_long, by = "value") |> 
    pull({{to_column}}) |> 
    replace_na(missing_value)
  
}
